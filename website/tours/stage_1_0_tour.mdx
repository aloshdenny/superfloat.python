```python
"""A tour of Stage 1.0.  Run: python examples/stage_1_0_tour.py"""
import warnings

import sf

print("formats:", [sf.info(d).name for d in sf.dtypes])
print(sf.info(sf.sf16))

a, b = sf.sf16(0.5), sf.sf16(0.25)
print("a + b =", a + b, "| a * b =", a * b, "| b / a =", b / a)
print("codes:", a.code, "bits:", hex(a.bits), "sign/mantissa:", a.sign, a.mantissa)
print("comparisons are exact:", a < 1, sf.sf16(0.1) == 0.1, sf.sf16(0.1) > 0.1)

for label, fn in [
    ("a / b (2.0)", lambda: a / b),
    ("a / 0", lambda: a / sf.sf16(0)),
    ("sf16(1)", lambda: sf.sf16(1)),
    ("a + 1 (literal)", lambda: a + 1),
    ("sqrt(-0.25)", lambda: sf.sqrt(sf.sf16(-0.25))),
    ("sf16(nan)", lambda: sf.sf16(float("nan"))),
    ("a // b", lambda: a // b),
    ("sf8(a) narrowing", lambda: sf.sf8(a)),
]:
    try:
        fn()
    except sf.SFError as e:
        print(f"{label:18s} -> {type(e).__name__}: {e}")

with warnings.catch_warnings(record=True) as w:
    warnings.simplefilter("always")
    print("0.005 * 0.005 =", sf.sf16(0.005) * 0.005, "with", w[0].category.__name__)

with sf.errstate(over="ignore"):
    print("saturation, like the GPU:", sf.sf16(0.75) + 0.5)
    x = sf.sf16(0.9)
    print("(0.9 + 0.9) - 0.9 =", (x + x) - x, "| 0.9 + (0.9 - 0.9) =", x + (x - x))

print("fma == mul then add:", sf.fma(a, b, 0.1).code == (a * b + 0.1).code)
print("promotion:", type(sf.sf8(0.25) + a).__name__, "| cast:", sf.cast(sf.sf16(0.1), sf.sf8))
print("policy:", sf.geterr())

```
